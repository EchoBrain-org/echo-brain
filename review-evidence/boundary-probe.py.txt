# Run from the review worktree after npm ci:
# python3 review-evidence/boundary-probe.py.txt
# Only a temporary archive is changed. It is removed in finally.
from pathlib import Path
import subprocess, tempfile, tarfile, io, json, shutil

root = Path(tempfile.mkdtemp(prefix='echo-boundary-review-probe-'))
try:
    data = subprocess.check_output(['git', 'archive', '3b663a74ec5d64e2f99b80ca590429df1415e02d'])
    with tarfile.open(fileobj=io.BytesIO(data)) as archive:
        archive.extractall(root)
    subprocess.run(['git', 'init', '-q', str(root)], check=True)
    subprocess.run(['git', '-C', str(root), 'add', '.'], check=True)
    checker = str(Path('tools/check-architecture-boundaries.mjs').resolve())
    def check(label):
        result = subprocess.run(['node', checker], cwd=root, text=True, capture_output=True)
        value = json.loads(result.stdout)
        print(json.dumps({'probe': label, 'ok': value['ok'], 'errors': value['errors']}))
    check('unchanged-main')
    manifest = root / 'product/source-boundary.v1.json'
    value = json.loads(manifest.read_text())
    value['adapter_architecture']['provider_neutral_paths'] += [
        'services/organization-authority/src/shared/core-runtime-observation-v1.ts',
        'packages/organization-control-plane/src/application/private-approval-policy-resolution-v1.ts',
        'packages/organization-api/src/person-tools.ts',
    ]
    manifest.write_text(json.dumps(value))
    check('classify-shared-policy-api-observation-as-neutral')
finally:
    shutil.rmtree(root)
